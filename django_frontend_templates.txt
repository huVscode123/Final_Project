<script>
let selectedSessionId = null;
const msgs = document.getElementById('chatMsgs');

function selectSession(sid, label) {
  selectedSessionId = sid;
  document.getElementById('selectedCtx').textContent = sid ? `Session #${sid} · ${label}` : 'None selected';
  document.querySelectorAll('.session-chip').forEach(c => c.classList.remove('selected'));
  document.getElementById(sid ? 'chip-' + sid : 'chip-none').classList.add('selected');
}

function quickAsk(text) {
  document.getElementById('userInput').value = text;
  sendMessage();
}

function appendMsg(cls, html, label='') {
  const div = document.createElement('div');
  div.className = cls;
  if (label) div.innerHTML = `<div class="msg-label">${label}</div>${html}`;
  else div.textContent = html;
  msgs.appendChild(div);
  msgs.scrollTop = msgs.scrollHeight;
  return div;
}

async function sendMessage() {
  const input = document.getElementById('userInput');
  const btn   = document.getElementById('sendBtn');
  const text  = input.value.trim();
  if (!text) return;

  appendMsg('msg-user', text);
  input.value = '';
  input.style.height = 'auto';
  btn.disabled = true;

  const thinking = appendMsg('msg-ai msg-thinking', '⟳ Analyzing...', '▸ VNAAP AI');

  const csrf = document.cookie.match(/csrftoken=([^;]+)/)?.[1] || '';
  try {
    const r = await fetch('/api/v1/ai/chat/', {
      method: 'POST',
      headers: { 'Content-Type':'application/json', 'X-CSRFToken': csrf },
      body: JSON.stringify({ message: text, session_id: selectedSessionId }),
    });
    const d = await r.json();
    thinking.remove();
    const reply = d.reply || d.error || '⚠ 無回應';
    appendMsg('msg-ai', reply, '▸ VNAAP AI');
  } catch(e) {
    thinking.remove();
    appendMsg('msg-ai', '⚠ 無法連線至 AI 服務（請確認 n8n webhook 已啟動）', '▸ SYSTEM');
    document.getElementById('backendStatus').textContent = 'Offline';
    document.getElementById('backendStatus').style.color = 'var(--red)';
  } finally {
    btn.disabled = false;
    msgs.scrollTop = msgs.scrollHeight;
  }
}

/* Auto-resize textarea */
document.getElementById('userInput').addEventListener('input', function() {
  this.style.height = 'auto';
  this.style.height = Math.min(this.scrollHeight, 120) + 'px';
});

    const r = await fetch('/api/v1/ai/chat/', {

    const d = await r.json();

    document.getElementById('backendStatus').textContent = 'Offline';
    document.getElementById('backendStatus').style.color = 'var(--text-muted)';
  }
})();

{% block extra_js %}
<script>
const SESSION_IDS = [{% for s in active_sessions %}{{ s.pk }},{% endfor %}];
const statusMap   = {pending:'等待中',running:'分析中',done:'已完成',failed:'分析失敗'};

async function pollActive() {
  document.getElementById('lastUpdate').textContent =
    'Updated: ' + new Date().toLocaleTimeString('zh-TW',{hour12:false});
  for (const sid of SESSION_IDS) {
    try {
      const r = await fetch(`/analyzer/sessions/${sid}/status/`);
      const d = await r.json();
      const st = document.getElementById('status-' + sid);
      if (st) {
        st.className = 'status-pill status-' + d.task_status;
        st.innerHTML = '<span class="dot"></span>' + (statusMap[d.task_status] || d.task_status);
      }
      const pk = document.getElementById('pkt-' + sid);
      const al = document.getElementById('alrt-' + sid);
      if (pk) pk.textContent = d.packet_count;
      if (al) al.textContent = d.alert_count;
      if (d.task_status === 'done' || d.task_status === 'failed') {
        setTimeout(() => location.reload(), 1200);
      }
    } catch(e) {}
  }
}

function loadLive() { location.reload(); }

if (SESSION_IDS.length > 0) {
  setInterval(pollActive, 5000);
  pollActive();
}

document.getElementById('lastUpdate').textContent =
  'Updated: ' + new Date().toLocaleTimeString('zh-TW',{hour12:false});
</script>
{% endblock %}

let pollTimer = null;
async function pollStatus() {

    const r = await fetch('{% url "analyzer:session_status" session.pk %}');
    const d = await r.json();

    document.getElementById('alertSpan').textContent = d.alert_count + ' alerts';
    if (d.task_status === 'done' || d.task_status === 'failed') {
      clearInterval(pollTimer);

    }
  } catch(e) {}
}

/* GradCAM trigger */
async function triggerGradCAM() {
  const btn  = document.getElementById('gcBtn');
  const msg  = document.getElementById('gcMsg');
  const csrf = document.cookie.match(/csrftoken=([^;]+)/)?.[1] || '';
  btn.disabled = true; btn.textContent = '⏳ Queuing...';
  msg.textContent = '';
  try {
    const r = await fetch('{% url "analyzer:trigger_gradcam" session.pk %}', {
      method: 'POST',
      headers: { 'X-CSRFToken': csrf, 'Content-Type': 'application/x-www-form-urlencoded' },
      body: new URLSearchParams({
        variant:      document.getElementById('gcVariant').value,
        max_images:   document.getElementById('gcMaxImgs').value,
        anomaly_only: document.getElementById('gcAnomalyOnly').checked ? 'true' : 'false',
        csrfmiddlewaretoken: csrf,
      }),
    });
    const d = await r.json();
    if (d.status === 'queued') {
      msg.style.color = '#22c55e';
      msg.textContent = '✓ ' + d.message;
    } else {
      msg.style.color = '#ef4444';
      msg.textContent = '✗ ' + (d.error || 'Error');
    }
  } catch(e) { msg.style.color='#ef4444'; msg.textContent = '✗ Connection failed'; }
  finally {
    btn.disabled = false;
    btn.innerHTML = '<svg fill="none" viewBox="0 0 24 24" stroke="currentColor" stroke-width="2" style="width:13px;height:13px"><polygon points="5 3 19 12 5 21 5 3"/></svg> Run Grad-CAM';
  }
}

/* File select */
const fileInput = document.getElementById('pcap_file');
const nameEl    = document.getElementById('selectedName');
const dropZone  = document.getElementById('dropZone');

fileInput.addEventListener('change', function() {
  const f = this.files[0];
  if (f) {
    nameEl.textContent = `✓ ${f.name} (${(f.size/1024/1024).toFixed(1)} MB)`;
    nameEl.style.display = 'block';
    dropZone.style.borderColor = 'var(--green)';
  }
});

/* Drag and drop */
['dragenter','dragover'].forEach(e => dropZone.addEventListener(e, ev => {
  ev.preventDefault(); dropZone.classList.add('drag-over');
}));
['dragleave','drop'].forEach(e => dropZone.addEventListener(e, ev => {
  ev.preventDefault(); dropZone.classList.remove('drag-over');
}));
dropZone.addEventListener('drop', ev => {
  const file = ev.dataTransfer.files[0];
  if (file) {
    const dt = new DataTransfer(); dt.items.add(file);
    fileInput.files = dt.files;
    fileInput.dispatchEvent(new Event('change'));
  }
});

/* XHR upload with progress */
document.getElementById('uploadForm').addEventListener('submit', function(e) {
  if (!fileInput.files[0]) return;
  e.preventDefault();
  const btn  = document.getElementById('submitBtn');
  const prog = document.getElementById('uploadProgress');
  const fill = document.getElementById('progressFill');
  const pct  = document.getElementById('pctText');
  btn.disabled = true; btn.textContent = '⏳ Uploading...';
  prog.style.display = 'block';

  const xhr = new XMLHttpRequest();
  xhr.upload.addEventListener('progress', ev => {
    if (ev.lengthComputable) {
      const p = Math.round(ev.loaded / ev.total * 100);
      fill.style.width = p + '%'; pct.textContent = p + '%';
    }
  });
  xhr.addEventListener('load', () => {
    fill.style.width = '100%'; pct.textContent = '100%';
    window.location.href = xhr.responseURL;

  const el = document.getElementById('headerClock');
  if (el) {
    const n = new Date();
    el.textContent = n.toLocaleTimeString('zh-TW', {hour12: false}) + ' UTC+8';
  }
  setTimeout(tick, 1000);
})();

  const w = document.getElementById('toastWrap');

}, 4000);
</script>
