async function changeRole(userId, newRole) {
  if (!await confirmAction(`確定要變更此使用者的角色為「${newRole}」嗎？`, '確認變更', 'primary')) {
    location.reload(); return;
  }
  try {
    const res = await VNAAP.api(`/accounts/admin/user/${userId}/`, {
      method: 'POST', body: JSON.stringify({ action: 'change_role', role: newRole })
    });
    showToast('角色已更新', 'success');
  } catch(e) { showToast('操作失敗', 'error'); location.reload(); }
}
async function toggleActive(userId) {
  if (!await confirmAction('確定要切換此使用者的啟用狀態嗎？', '確認', 'danger')) return;
  try {
    const res = await VNAAP.api(`/accounts/admin/user/${userId}/`, {
      method: 'POST', body: JSON.stringify({ action: 'toggle_active' })
    });
    showToast('狀態已更新', 'success');
    location.reload();
  } catch(e) { showToast('操作失敗', 'error'); }
}

<script>
document.addEventListener('DOMContentLoaded', function() {
    const btnRun = document.getElementById('btn-run');
    const btnRefresh = document.getElementById('btn-refresh-log');
    const terminalOutput = document.getElementById('terminal-output');
    const statusBadge = document.getElementById('current-status');
    const resultsPanel = document.getElementById('results-panel');
    const resultsGrid = document.getElementById('results-grid');
    const datasetSelect = document.getElementById('dataset-select');
    
    let pollInterval = null;
    
    // 全選/取消全選邏輯
    const allCheckbox = document.querySelector('input[value="all"]');
    const expCheckboxes = document.querySelectorAll('input[name="experiments"]:not([value="all"])');
    
    allCheckbox.addEventListener('change', function() {
        expCheckboxes.forEach(cb => {
            cb.checked = false;
            cb.disabled = this.checked;
        });
    });
    
    expCheckboxes.forEach(cb => {
        cb.addEventListener('change', function() {
            if (this.checked) {
                allCheckbox.checked = false;
            }
        });
    });
    
    // 初始化 Checkbox 狀態
    expCheckboxes.forEach(cb => cb.disabled = true);
    
    function updateStatusUI(status) {
        statusBadge.className = 'status-badge';
        if (status === 'running') {
            statusBadge.classList.add('status-running');
            statusBadge.innerHTML = '<span class="spinner"></span> 正在執行中...';
            btnRun.disabled = true;
        } else if (status === 'done') {
            statusBadge.classList.add('status-done');
            statusBadge.innerHTML = '<i class="fas fa-check-circle"></i> 實驗完成';
            btnRun.disabled = false;
        } else if (status === 'error') {
            statusBadge.classList.add('status-error');
            statusBadge.innerHTML = '<i class="fas fa-exclamation-triangle"></i> 執行失敗';
            btnRun.disabled = false;
        } else {
            statusBadge.classList.add('status-idle');
            statusBadge.innerHTML = '<i class="fas fa-pause-circle"></i> 尚未執行';
            btnRun.disabled = false;
        }
    }
    
    function renderImages(images) {
        if (!images || Object.keys(images).length === 0) return;
        
        resultsPanel.style.display = 'block';
        resultsGrid.innerHTML = '';
        
        for (const [filename, url] of Object.entries(images)) {
            const title = filename.replace('ablation_', '').replace('.png', '').replace(/_/g, ' ').toUpperCase();
            
            const card = document.createElement('div');
            card.className = 'result-card fade-in';
            card.innerHTML = `
                <h4>${title}</h4>
                <a href="${url}" target="_blank">
                    <img src="${url}" class="result-image" alt="${title}">
                </a>
            `;
            resultsGrid.appendChild(card);
        }
    }
    
    function fetchStatus() {
        const dataset = datasetSelect.value;
        fetch(`/analyzer/ablation/api/status/?dataset=${dataset}`)
            .then(res => res.json())
            .then(data => {
                if (data.ok) {
                    updateStatusUI(data.status);
                    
                    if (data.logs) {
                        terminalOutput.textContent = data.logs;
                        terminalOutput.scrollTop = terminalOutput.scrollHeight;
                    }
                    
                    if (data.status === 'done' || data.status === 'error') {
                        clearInterval(pollInterval);
                        pollInterval = null;
                        
                        if (data.status === 'done' && data.images) {
                            renderImages(data.images);
                        }
                    }
                }
            })
            .catch(err => console.error('Error fetching status:', err));
    }
    
    btnRun.addEventListener('click', function() {
        const dataset = datasetSelect.value;
        let experiments = [];
        
        if (allCheckbox.checked) {
            experiments = ['all'];
        } else {
            expCheckboxes.forEach(cb => {
                if (cb.checked) experiments.push(cb.value);
            });
            if (experiments.length === 0) {
                alert('請至少選擇一項實驗');
                return;
            }
        }
        
        // 重置 UI
        terminalOutput.textContent = '啟動中...';
        resultsPanel.style.display = 'none';
        updateStatusUI('running');
        
        // 發送執行請求
        fetch('/analyzer/ablation/api/run/', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'X-CSRFToken': VNAAP.getCookie('csrftoken')
            },
            body: JSON.stringify({
                dataset: dataset,
                experiments: experiments
            })
        })
        .then(res => res.json())
        .then(data => {
            if (data.ok) {
                // 開始輪詢
                if (pollInterval) clearInterval(pollInterval);
                pollInterval = setInterval(fetchStatus, 2000);
            } else {
                alert('啟動失敗: ' + data.error);
                updateStatusUI('error');
            }
        })
        .catch(err => {
            console.error('Run API Error:', err);
            alert('系統錯誤');
            updateStatusUI('error');
        });
    });
    
    btnRefresh.addEventListener('click', fetchStatus);
    
    // 進入頁面時先抓一次狀態
    fetchStatus();
});
</script>

function renderQuickButtons(mode) {
  const list = document.getElementById('quickBtnList');

  MODE_CONFIG[mode].quick.forEach(q => {
    const btn = document.createElement('button');

function selectMode(mode, btnEl) {
  currentMode = mode;
  document.querySelectorAll('.mode-btn').forEach(b => b.classList.remove('active'));
  btnEl.classList.add('active');
  document.getElementById('modeHint').textContent = MODE_CONFIG[mode].hint;

function selectSession(sid, label) {
  selectedSessionId = sid;
  document.getElementById('selectedCtx').textContent = sid ? `Session #${sid} · ${label}` : 'None selected';
  document.querySelectorAll('.session-chip').forEach(c => c.classList.remove('selected'));

  document.getElementById('userInput').value = text;
  sendMessage();
}

function appendMsg(cls, html, label='') {
  const div = document.createElement('div');
  div.className = cls;
  if (label) div.innerHTML = `<div class="msg-label">${label}</div>${html}`;
  else div.textContent = html;
  msgs.appendChild(div);
  msgs.scrollTop = msgs.scrollHeight;
  return div;
}

async function sendMessage() {
  const input = document.getElementById('userInput');
  const btn   = document.getElementById('sendBtn');
  const text  = input.value.trim();
  if (!text) return;

  appendMsg('msg-user', text);
  input.value = '';
  input.style.height = 'auto';
  btn.disabled = true;

  const thinking = appendMsg('msg-ai msg-thinking', '⟳ 思考中...', '▸ VNAAP AI');

  const csrf = document.cookie.match(/csrftoken=([^;]+)/)?.[1] || '';
  try {
    const r = await fetch('/api/v1/ai/chat/', {
      method: 'POST',
      headers: { 'Content-Type':'application/json', 'X-CSRFToken': csrf },
      body: JSON.stringify({
        message: text,
        mode: currentMode,
        session_id: selectedSessionId,
      }),
    });
    const d = await r.json();
    thinking.remove();
    const reply = d.reply || d.error || '⚠ 無回應';
    appendMsg('msg-ai', reply, '▸ VNAAP AI · ' + MODE_CONFIG[currentMode].label);
  } catch(e) {
    thinking.remove();
    appendMsg('msg-ai', '⚠ 無法連線至 AI 服務（請確認 n8n webhook 已啟動）', '▸ SYSTEM');
    document.getElementById('backendStatus').textContent = 'Offline';
    document.getElementById('backendStatus').style.color = 'var(--red)';
  } finally {
    btn.disabled = false;
    msgs.scrollTop = msgs.scrollHeight;
  }
}

<script>
const SESSION_IDS = [{% for s in active_sessions %}{{ s.pk }},{% endfor %}];
const statusMap   = {pending:'等待中',running:'分析中',done:'已完成',failed:'分析失敗'};

async function pollActive() {
  document.getElementById('lastUpdate').textContent =
    'Updated: ' + new Date().toLocaleTimeString('zh-TW',{hour12:false});
  for (const sid of SESSION_IDS) {
    try {
      const r = await fetch(`/analyzer/sessions/${sid}/status/`);
      const d = await r.json();
      const st = document.getElementById('status-' + sid);
      if (st) {
        st.className = 'status-pill status-' + d.task_status;
        st.innerHTML = '<span class="dot"></span>' + (statusMap[d.task_status] || d.task_status);
      }
      const pk = document.getElementById('pkt-' + sid);
      const al = document.getElementById('alrt-' + sid);
      if (pk) pk.textContent = d.packet_count;
      if (al) al.textContent = d.alert_count;
      if (d.task_status === 'done' || d.task_status === 'failed') {
        setTimeout(() => location.reload(), 1200);
      }
    } catch(e) {}
  }
}

function loadLive() { location.reload(); }

if (SESSION_IDS.length > 0) {
  setInterval(pollActive, 5000);
  pollActive();
}

document.getElementById('lastUpdate').textContent =
  'Updated: ' + new Date().toLocaleTimeString('zh-TW',{hour12:false});
</script>

(function() {
  const normal = {{ cnn_result.avg_normal_error|default:0 }};
  const anomaly = {{ cnn_result.avg_attack_error|default:0 }};
  const thresh = {{ cnn_result.threshold|default:0 }};
  const normalCount = {{ cnn_result.normal_count|default:0 }};
  const anomalyCount = {{ cnn_result.anomaly_count|default:0 }};

  // 產生模擬分布資料
  const bins = 20;
  const maxErr = Math.max(anomaly * 2, thresh * 3, 0.001);
  const step = maxErr / bins;
  const normalData = [], anomalyData = [], labels = [];

  for (let i = 0; i < bins; i++) {
    const center = step * (i + 0.5);
    labels.push(center.toFixed(5));
    // 使用高斯分布模擬
    const nVal = normalCount * Math.exp(-0.5 * Math.pow((center - normal) / (normal * 0.3 + 1e-6), 2));
    const aVal = anomalyCount * Math.exp(-0.5 * Math.pow((center - anomaly) / (anomaly * 0.4 + 1e-6), 2));
    normalData.push(Math.max(0, nVal));
    anomalyData.push(Math.max(0, aVal));
  }

  const ctx = document.getElementById('cnnErrorChart');
  if (ctx) {
    new Chart(ctx, {
      type: 'bar',
      data: {
        labels: labels,
        datasets: [
          { label: '正常封包', data: normalData,

let pollTimer = null;
async function pollStatus() {
  try {
    const r = await fetch('{% url "analyzer:session_status" session.pk %}');
    const d = await r.json();
    const m = { pending:'等待中', running:'分析中', done:'已完成', failed:'分析失敗' };
    const badge = document.getElementById('statusBadge');
    badge.className = 'status-pill status-' + d.task_status;
    badge.innerHTML = '<span class="dot"></span>' + (m[d.task_status] || d.task_status);
    document.getElementById('pktSpan').textContent   = d.packet_count + ' pkts';
    document.getElementById('alertSpan').textContent = d.alert_count + ' alerts';
    if (d.task_status === 'done' || d.task_status === 'failed') {
      clearInterval(pollTimer);
      if (d.task_status === 'done') setTimeout(() => location.reload(), 800);
    }
  } catch(e) {}
}

/* GradCAM trigger */
async function triggerGradCAM() {
  const btn  = document.getElementById('gcBtn');
  const msg  = document.getElementById('gcMsg');
  const csrf = document.cookie.match(/csrftoken=([^;]+)/)?.[1] || '';
  btn.disabled = true; btn.textContent = '⏳ Queuing...';
  msg.textContent = '';
  try {
    const r = await fetch('{% url "analyzer:trigger_gradcam" session.pk %}', {
      method: 'POST',
      headers: { 'X-CSRFToken': csrf, 'Content-Type': 'application/x-www-form-urlencoded' },
      body: new URLSearchParams({
        variant:      document.getElementById('gcVariant').value,
        max_images:   document.getElementById('gcMaxImgs').value,
        anomaly_only: document.getElementById('gcAnomalyOnly').checked ? 'true' : 'false',
        csrfmiddlewaretoken: csrf,
      }),
    });
    const d = await r.json();
    if (d.status === 'queued') {
      msg.style.color = '#22c55e';
      msg.textContent = '✓ ' + d.message;
    } else {
      msg.style.color = '#ef4444';
      msg.textContent = '✗ ' + (d.error || 'Error');
    }
  } catch(e) { msg.style.color='#ef4444'; msg.textContent = '✗ Connection failed'; }
  finally {
    btn.disabled = false;
    btn.innerHTML = '<svg fill="none" viewBox="0 0 24 24" stroke="currentColor" stroke-width="2" style="width:13px;height:13px"><polygon points="5 3 19 12 5 21 5 3"/></svg> Run Grad-CAM';
  }
}

    const tbody = document.getElementById('resultTbody');
    tbody.innerHTML = '';
    res.results.forEach((r, i) => {
      const tr = document.createElement('tr');
      tr.className = 'pkt-row';
      tr.style.animationDelay = `${Math.min(i, 40) * 0.02}s`;

        <td class="mono">${r.index + 1}</td>
        <td>${VNAAP.formatBytes(r.size)}</td>

    });

    showToast(`檢測完成：直接觸發 ${res.anomaly_count} 筆；批次判定為 ${res.rule_based_anomaly_detected ? '攻擊' : '正常'}`, res.rule_based_anomaly_detected ? 'warning' : 'success');
  } catch(e) {
    document.getElementById('loadingResult').style.display = 'none';
    document.getElementById('emptyResult').style.display = 'block';
    const errMsg = (e && e.data && e.data.error) ? e.data.error : (e && e.message ? e.message : '未知錯誤');
    showToast('模擬檢測失敗：' + errMsg, 'error');
  } finally {
    btn.disabled = false;
  }
}

function sleep(ms) { return new Promise(r => setTimeout(r, ms)); }

function updateSteps(current) {
  for (let i = 1; i <= 3; i++) {
    const el = document.getElementById(`step${i}`);
    if (!el) continue;
    el.className = 'sim-step ' + (i < current ? 'done' : i === current ? 'active' : 'pending');
  }
  const texts = ['', '產生模擬封包中...', 'CNN Autoencoder 推論中...', '分析結果生成中...'];
  const loadText = document.getElementById('loadingText');
  if (loadText) loadText.textContent = texts[current] || '';
}

function animateValue(id, target) {
  const el = document.getElementById(id);
  if (!el) return;
  let current = 0;
  const step = Math.max(1, Math.ceil(target / 30));
  const interval = setInterval(() => {
    current += step;
    if (current >= target) { current = target; clearInterval(interval); }
    el.textContent = current;
  }, 30);
}

function renderScoreChart(results, threshold, baseline) {
  const ctx = document.getElementById('scoreChart').getContext('2d');
  if (scoreChartInstance) scoreChartInstance.destroy();

  const datasets = [{
    label: '本次模擬 CNN 分數（顯示樣本）',
    data: results.map(r => r.score),
    backgroundColor: results.map(r => r.cnn_anomaly ? 'rgba(234,179,8,0.75)' : 'rgba(59,130,246,0.55)'),
    borderColor: results.map(r => r.cnn_anomaly ? '#eab308' : '#3b82f6'),
    borderWidth: 1, borderRadius: 3, order: 2,
  }];

  if (baseline && baseline.scores && baseline.scores.length) {
    datasets.push({
      type: 'line',
      label: '基準線 (正常流量取樣)',
      data: results.map((_, i) => baseline.scores[i % baseline.scores.length]),
      borderColor: 'rgba(148,163,184,0.9)',
      backgroundColor: 'rgba(148,163,184,0.9)',
      borderDash: [3, 3], borderWidth: 1.5,
      pointRadius: 2, pointStyle: 'circle',
      fill: false, tension: 0, order: 1,
    });
  }

  scoreChartInstance = new Chart(ctx, {
    type: 'bar',
    data: {
      labels: results.map(r => `#${r.index + 1}`),
      datasets,
    },
    options: {
      responsive: true, maintainAspectRatio: false,
      plugins: {
        legend: { display: !!(baseline && baseline.scores && baseline.scores.length), labels: { color: '#94a3b8', font: { size: 10 } } },
        annotation: { annotations: {
          threshold: { type: 'line', yMin: threshold, yMax: threshold, borderColor: '#f97316', borderWidth: 2, borderDash: [5,5], label: { display: true, content: `判定閾值 ${threshold}`, position: 'end', backgroundColor: 'rgba(249,115,22,0.8)', font: { size: 10 } } }
        }}
      },
      scales: {
        x: { ticks: { color: '#475569', font: { size: 10 } }, grid: { display: false } },
        y: { ticks: { color: '#475569', font: { size: 10 } }, grid: { color: 'rgba(255,255,255,0.04)' }, beginAtZero: true }
      }
    }
  });
}
