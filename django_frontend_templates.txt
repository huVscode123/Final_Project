async function changeRole(userId, newRole) {
  if (!await confirmAction(`確定要變更此使用者的角色為「${newRole}」嗎？`, '確認變更', 'primary')) {
    location.reload(); return;
  }
  try {
    const res = await VNAAP.api(`/accounts/admin/user/${userId}/`, {
      method: 'POST', body: JSON.stringify({ action: 'change_role', role: newRole })
    });
    showToast('角色已更新', 'success');
  } catch(e) { showToast('操作失敗', 'error'); location.reload(); }
}
async function toggleActive(userId) {
  if (!await confirmAction('確定要切換此使用者的啟用狀態嗎？', '確認', 'danger')) return;
  try {
    const res = await VNAAP.api(`/accounts/admin/user/${userId}/`, {
      method: 'POST', body: JSON.stringify({ action: 'toggle_active' })
    });
    showToast('狀態已更新', 'success');
    location.reload();
  } catch(e) { showToast('操作失敗', 'error'); }
}

<script>
document.addEventListener('DOMContentLoaded', function() {
    const btnRun = document.getElementById('btn-run');
    const btnRefresh = document.getElementById('btn-refresh-log');
    const terminalOutput = document.getElementById('terminal-output');
    const statusBadge = document.getElementById('current-status');
    const resultsPanel = document.getElementById('results-panel');
    const resultsGrid = document.getElementById('results-grid');
    const datasetSelect = document.getElementById('dataset-select');

    let pollInterval = null;

    // 全選/取消全選邏輯
    const allCheckbox = document.querySelector('input[value="all"]');
    const expCheckboxes = document.querySelectorAll('input[name="experiments"]:not([value="all"])');

    allCheckbox.addEventListener('change', function() {
        expCheckboxes.forEach(cb => {
            cb.checked = false;
            cb.disabled = this.checked;
        });
    });

    expCheckboxes.forEach(cb => {
        cb.addEventListener('change', function() {
            if (this.checked) {
                allCheckbox.checked = false;
            }
        });
    });

    // 初始化 Checkbox 狀態
    expCheckboxes.forEach(cb => cb.disabled = true);

    function updateStatusUI(status) {
        statusBadge.className = 'status-badge';
        if (status === 'running') {
            statusBadge.classList.add('status-running');
            statusBadge.innerHTML = '<span class="spinner"></span> 正在執行中...';
            btnRun.disabled = true;
        } else if (status === 'done') {
            statusBadge.classList.add('status-done');
            statusBadge.innerHTML = '<i class="fas fa-check-circle"></i> 實驗完成';
            btnRun.disabled = false;
        } else if (status === 'error') {
            statusBadge.classList.add('status-error');
            statusBadge.innerHTML = '<i class="fas fa-exclamation-triangle"></i> 執行失敗';
            btnRun.disabled = false;
        } else {
            statusBadge.classList.add('status-idle');
            statusBadge.innerHTML = '<i class="fas fa-pause-circle"></i> 尚未執行';
            btnRun.disabled = false;
        }
    }

    function renderImages(images) {
        if (!images || Object.keys(images).length === 0) return;

        resultsPanel.style.display = 'block';
        resultsGrid.innerHTML = '';

        for (const [filename, url] of Object.entries(images)) {
            const title = filename.replace('ablation_', '').replace('.png', '').replace(/_/g, ' ').toUpperCase();

            const card = document.createElement('div');
            card.className = 'result-card fade-in';
            card.innerHTML = `
                <h4>${title}</h4>
                <a href="${url}" target="_blank">
                    <img src="${url}" class="result-image" alt="${title}">
                </a>
            `;
            resultsGrid.appendChild(card);
        }
    }

    function fetchStatus() {
        const dataset = datasetSelect.value;
        fetch(`/analyzer/ablation/api/status/?dataset=${dataset}`)
            .then(res => res.json())
            .then(data => {
                if (data.ok) {
                    updateStatusUI(data.status);

                    if (data.logs) {
                        terminalOutput.textContent = data.logs;
                        terminalOutput.scrollTop = terminalOutput.scrollHeight;
                    }

                    if (data.status === 'done' || data.status === 'error') {
                        clearInterval(pollInterval);
                        pollInterval = null;

                        if (data.status === 'done' && data.images) {
                            renderImages(data.images);
                        }
                    }
                }
            })
            .catch(err => console.error('Error fetching status:', err));
    }

    btnRun.addEventListener('click', function() {
        const dataset = datasetSelect.value;
        let experiments = [];

        if (allCheckbox.checked) {
            experiments = ['all'];
        } else {
            expCheckboxes.forEach(cb => {
                if (cb.checked) experiments.push(cb.value);
            });
            if (experiments.length === 0) {
                alert('請至少選擇一項實驗');
                return;
            }
        }

        // 重置 UI

const SESSION_IDS = [{% for s in active_sessions %}{{ s.pk }},{% endfor %}];
const statusMap   = {pending:'等待中',running:'分析中',done:'已完成',failed:'分析失敗'};

async function pollActive() {
  document.getElementById('lastUpdate').textContent =
    'Updated: ' + new Date().toLocaleTimeString('zh-TW',{hour12:false});
  for (const sid of SESSION_IDS) {
    try {
      const r = await fetch(`/analyzer/sessions/${sid}/status/`);
      const d = await r.json();
      const st = document.getElementById('status-' + sid);
      if (st) {
        st.className = 'status-pill status-' + d.task_status;

      if (pk) pk.textContent = d.packet_count;
      if (al) al.textContent = d.alert_count;
      if (d.task_status === 'done' || d.task_status === 'failed') {
        setTimeout(() => location.reload(), 1200);
      }
    } catch(e) {}
  }
}

function loadLive() { location.reload(); }

if (SESSION_IDS.length > 0) {
  setInterval(pollActive, 5000);
  pollActive();
}

document.getElementById('lastUpdate').textContent =

(function() {
  const normal = {{ cnn_result.avg_normal_error|default:0 }};
  const anomaly = {{ cnn_result.avg_attack_error|default:0 }};
  const thresh = {{ cnn_result.threshold|default:0 }};
  const normalCount = {{ cnn_result.normal_count|default:0 }};
  const anomalyCount = {{ cnn_result.anomaly_count|default:0 }};

  // 產生模擬分布資料
  const bins = 20;
  const maxErr = Math.max(anomaly * 2, thresh * 3, 0.001);
  const step = maxErr / bins;
  const normalData = [], anomalyData = [], labels = [];

  for (let i = 0; i < bins; i++) {
    const center = step * (i + 0.5);
    labels.push(center.toFixed(5));
    // 使用高斯分布模擬
    const nVal = normalCount * Math.exp(-0.5 * Math.pow((center - normal) / (normal * 0.3 + 1e-6), 2));
    const aVal = anomalyCount * Math.exp(-0.5 * Math.pow((center - anomaly) / (anomaly * 0.4 + 1e-6), 2));
    normalData.push(Math.max(0, nVal));
    anomalyData.push(Math.max(0, aVal));
  }

  const ctx = document.getElementById('cnnErrorChart');
  if (ctx) {
    new Chart(ctx, {
      type: 'bar',
      data: {
        labels: labels,
        datasets: [
          { label: '正常封包', data: normalData,

let pollTimer = null;
async function pollStatus() {
  try {
    const r = await fetch('{% url "analyzer:session_status" session.pk %}');
    const d = await r.json();
    const m = { pending:'等待中', running:'分析中', done:'已完成', failed:'分析失敗' };
    const badge = document.getElementById('statusBadge');
    badge.className = 'status-pill status-' + d.task_status;
    badge.innerHTML = '<span class="dot"></span>' + (m[d.task_status] || d.task_status);
    document.getElementById('pktSpan').textContent   = d.packet_count + ' pkts';
    document.getElementById('alertSpan').textContent = d.alert_count + ' alerts';
    if (d.task_status === 'done' || d.task_status === 'failed') {
      clearInterval(pollTimer);
      if (d.task_status === 'done') setTimeout(() => location.reload(), 800);
    }
  } catch(e) {}
}

/* GradCAM trigger */
async function triggerGradCAM() {
  const btn  = document.getElementById('gcBtn');
  const msg  = document.getElementById('gcMsg');
  const csrf = document.cookie.match(/csrftoken=([^;]+)/)?.[1] || '';
  btn.disabled = true; btn.textContent = '⏳ Queuing...';
  msg.textContent = '';
  try {
    const r = await fetch('{% url "analyzer:trigger_gradcam" session.pk %}', {
      method: 'POST',
      headers: { 'X-CSRFToken': csrf, 'Content-Type': 'application/x-www-form-urlencoded' },
      body: new URLSearchParams({
        variant:      document.getElementById('gcVariant').value,
        max_images:   document.getElementById('gcMaxImgs').value,
        anomaly_only: document.getElementById('gcAnomalyOnly').checked ? 'true' : 'false',
        csrfmiddlewaretoken: csrf,
      }),
    });
    const d = await r.json();
    if (d.status === 'queued') {
      msg.style.color = 'var(--green)';
      msg.textContent = '✓ ' + d.message;
    } else {
      msg.style.color = 'var(--red)';
      msg.textContent = '✗ ' + (d.error || 'Error');
    }
  } catch(e) { msg.style.color='var(--red)'; msg.textContent = '✗ Connection failed'; }
  finally {
    btn.disabled = false;
    btn.innerHTML = '<svg fill="none" viewBox="0 0 24 24" stroke="currentColor" stroke-width="2" style="width:13px;height:13px"><polygon points="5 3 19 12 5 21 5 3"/></svg> Run Grad-CAM';
  }
}

<script>
let selectedType = 'port_scan';
function selectAttack(el) {
  document.querySelectorAll('.attack-card').forEach(c => c.classList.remove('selected'));
  el.classList.add('selected');
  selectedType = el.dataset.type;
}

function setStep(n) {
  for (let i = 1; i <= 3; i++) {
    const el = document.getElementById('step' + i);
    el.className = 'sim-step' + (i < n ? ' done' : i === n ? ' active' : '');
  }
}

async function runSimulation() {
  const btn = document.getElementById('runBtn');
  const content = document.getElementById('resultContent');
  const pktCount = document.getElementById('pktCount').value;
  btn.disabled = true;

  setStep(1);
  content.innerHTML = `<div class="result-box"><div class="spinner-lg"></div><div style="margin-top:14px;font-size:.85rem;color:var(--text-secondary)">正在產生合成封包資料...</div></div>`;
  await new Promise(r => setTimeout(r, 700));

  setStep(2);
  content.innerHTML = `<div class="result-box"><div class="spinner-lg"></div><div style="margin-top:14px;font-size:.85rem;color:var(--text-secondary)">CNN Autoencoder 分析中...</div></div>`;

  const csrf = document.cookie.match(/csrftoken=([^;]+)/)?.[1] || '';
  try {
    const r = await fetch('{% url "analyzer:simulation_api" %}', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'X-CSRFToken': csrf },
      body: JSON.stringify({ attack_type: selectedType, packet_count: parseInt(pktCount) }),
    });
    const d = await r.json();
    setStep(3);
    await new Promise(res => setTimeout(res, 300));

    if (d.error) {
      content.innerHTML = `<div class="result-box verdict-anomaly"><div class="verdict-badge">⚠ ERROR</div><p style="margin-top:10px;font-size:.8rem">${d.error}</p></div>`;
    } else {
      const isAnomaly = d.is_anomaly;
      content.innerHTML = `
        <div class="result-box ${isAnomaly ? 'verdict-anomaly' : 'verdict-normal'}">
          <div class="verdict-badge">${isAnomaly ? '🚨 ANOMALY DETECTED' : '✓ NORMAL TRAFFIC'}</div>
          <div style="display:grid;grid-template-columns:repeat(3,1fr);gap:12px;margin-top:18px;text-align:left">
            <div><div style="font-size:.66rem;color:var(--text-muted)">重建誤差</div><div class="mono" style="font-size:1rem;font-weight:700">${d.reconstruction_error?.toFixed(6) ?? '—'}</div></div>
            <div><div style="font-size:.66rem;color:var(--text-muted)">閾值</div><div class="mono" style="font-size:1rem;font-weight:700;color:var(--yellow)">${d.threshold?.toFixed(6) ?? '—'}</div></div>
            <div><div style="font-size:.66rem;color:var(--text-muted)">攻擊類型</div><div style="font-size:.85rem;font-weight:600">${d.attack_type_label ?? selectedType}</div></div>
          </div>
        </div>
        <div class="card" style="margin-top:14px;padding:14px">
          <div style="font-size:.8rem;font-weight:600;margin-bottom:8px">模擬封包摘要</div>
          <div style="font-size:.78rem;color:var(--text-muted);line-height:1.8">
            產生 ${d.packet_count ?? pktCount} 個合成封包 · 平均封包大小 ${d.avg_packet_size ?? '—'} bytes
          </div>
        </div>`;
    }
  } catch(e) {
    content.innerHTML = `<div class="result-box verdict-anomaly"><div class="verdict-badge">⚠ 連線失敗</div><p style="margin-top:10px;font-size:.8rem">無法連線至分析服務，請稍後再試。</p></div>`;
  } finally {
    btn.disabled = false;
  }
}
</script>
